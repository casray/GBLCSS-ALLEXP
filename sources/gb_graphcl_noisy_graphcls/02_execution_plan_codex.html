def augment_node_dropping(data, drop_ratio=0.2):

def nt_xent_loss(z1, z2, temperature=0.2):
    """NT-Xent (InfoNCE) loss。z1, z2 shape: [batch_size, dim]。
    正样本对 = 同图的两个视图。负样本 = batch 中其他图。
    返回标量 loss。"""

def train_graphcl_epoch(encoder, proj_head, loader, optimizer, device, aug1_fn, aug2_fn, temperature=0.2):
    """一个 epoch 的 GraphCL 训练。
    对每个 batch：
      view1 = aug1_fn(batch) → encoder → proj_head → z1
      view2 = aug2_fn(batch) → encoder → proj_head → z2
      loss = nt_xent_loss(z1, z2)
    返回 epoch mean loss。"""

2.4 src/gb_graphcl/granular_ball.py

def build_granular_balls(embeddings: Tensor, noisy_labels: Tensor, train_indices: list,
                         purity_threshold=0.7, min_samples=5, max_size=100):
    """在 train_indices 对应的 L2-normalized embeddings 上构建密度引导粒球。

    算法：
    1. 初始：所有 train 样本在一个球中。
    2. 递归分裂：
       - 计算球内 noisy_label 多数票占比 = purity
       - 如果 purity >= purity_threshold 且 len(ball) <= max_size → 停止
       - 如果 len(ball) < min_samples → 停止（不再分裂太小的球）
       - 否则：用 sklearn KMeans(n_clusters=2) 分成两个子球，递归
    3. 返回:
       - group_ids: Tensor[num_train], 每个 train 样本属于哪个球 (0, 1, ..., num_balls-1)
       - ball_centers: Tensor[num_balls, dim], 每个球的成员均值（L2 归一化后）
       - ball_sizes: list[int], 每个球的大小
       - ball_purities: list[float], 每个球的 noisy-label purity

    注意：只用 noisy_labels 计算 purity，绝不用 clean_labels。
    """

def shuffle_group_ids(group_ids: Tensor, seed: int) -> Tensor:
    """打乱 group_ids（保持每个组的大小不变），破坏语义结构。"""

def random_group_ids(num_samples: int, group_sizes: list, seed: int) -> Tensor:
    """随机分配成员到与 GB 同大小的组中。"""

2.5 src/gb_graphcl/reliability.py

def knn_consensus_reliability(embeddings: Tensor, noisy_labels: Tensor,
                               train_indices: list, k: int) -> Tensor:
    """对每个 train 样本，找 embedding 空间中 k 近邻（限 train），
    计算邻居中与该样本同 noisy label 的比例。
    返回 Tensor[num_train]，值域 [0, 1]。"""

def kmeans_consensus_reliability(embeddings: Tensor, noisy_labels: Tensor,
                                  train_indices: list, n_clusters: int) -> Tensor:
    """对 train embedding 做 kmeans 聚类（n_clusters 个 cluster），
    每个样本的 reliability = 其 cluster 内与该样本同 noisy label 的比例。
    返回 Tensor[num_train]。"""

def gb_consensus_reliability(group_ids: Tensor, noisy_labels: Tensor,
                              train_indices: list) -> Tensor:
    """每个样本的 reliability = 其所在球内与该样本同 noisy label 的比例。
    返回 Tensor[num_train]。"""

def compute_reliability(method: str, embeddings, noisy_labels, train_indices, **kwargs) -> Tensor:
    """统一入口。method in {"topk", "kmeans", "gb", "shuffled", "random", "none"}.
    - "topk": 调 knn_consensus_reliability, k=kwargs["k"]
    - "kmeans": 调 kmeans_consensus_reliability, n_clusters=kwargs["n_clusters"]
    - "gb": 调 gb_consensus_reliability, group_ids=kwargs["group_ids"]
    - "shuffled": 调 gb_consensus_reliability, group_ids=shuffle_group_ids(kwargs["group_ids"], kwargs["seed"])
    - "random": 调 gb_consensus_reliability, group_ids=random_group_ids(...)
    - "none": 返回全零（纯 small-loss，不用 reliability）
    """

2.6 src/gb_graphcl/selection.py

def keep_ratio_at_epoch(epoch: int, warmup: int = 20, ramp: int = 20, rho: float = 0.6) -> float:
    """epoch < warmup: return 1.0
    warmup <= epoch < warmup+ramp: 线性从 1.0 降到 rho
    epoch >= warmup+ramp: return rho"""

def train_with_selection(embeddings, noisy_labels, clean_labels, train_idx, val_idx, test_idx,
                          reliability, num_classes, device,
                          epochs=200, lr=0.01, wd=5e-4, patience=30,
                          warmup=20, ramp=20, rho=0.6, seed=0):
    """在 frozen embeddings 上训练线性分类器 + commit-style sample selection。

    每 epoch:
      1. 前向：logits = Linear(embeddings[train_idx])
      2. 算 per-sample CE loss
      3. trust_score = z_score(-loss) + z_score(reliability)   # reliability 是预计算的固定值
         如果 reliability 全零（sel_noball），trust_score = z_score(-loss)
      4. keep_ratio = keep_ratio_at_epoch(epoch, warmup, ramp, rho)
      5. 选 trust_score 前 keep_ratio 比例的样本
      6. 只对选中样本的 loss 求均值，反向传播
      7. val accuracy early stopping (patience)

    返回 dict:
      test_acc_at_best_val: float
      selection_precision: float  # 最后一个 epoch 选中样本中 clean 的比例（用 clean_labels 事后算）
      noise_exclusion_recall: float  # 最后一个 epoch noisy 样本被排除的比例
      loss_history: list
    """

3. 脚本实现规格

3.1 scripts/train_graphcl.py

用法: python scripts/train_graphcl.py --dataset NCI1 --seed 0

流程:
1. 加载数据集
2. 做 stratified split（8:1:1）
3. 只用 train+val 图（不用 test）做 GraphCL 训练
   - GIN 5层, hidden=128
   - aug1 = node_dropping(0.2), aug2 = edge_perturbation(0.2)
   - NT-Xent, temperature=0.2
   - Adam lr=0.001, 100 epochs, batch_size=128
4. 保存 encoder 权重到 checkpoints/{dataset}/pretrain{seed}/encoder.pt
5. 打印最终 loss

3.2 scripts/extract_embeddings.py

用法: python scripts/extract_embeddings.py --dataset NCI1 --seed 0

流程:
1. 加载 encoder 权重
2. 对所有图提取 embedding（不含 projection head）
3. L2 归一化
4. 保存到 checkpoints/embeddings/{dataset}/pretrain{seed}/H.pt
   格式: {"embeddings": Tensor[N, 128], "labels": Tensor[N], "num_classes": int}

3.3 scripts/run_smoke.py

用法: python scripts/run_smoke.py --dataset NCI1 --noise-type symmetric --noise-rate 0.4 --seed 0

流程:
1. 加载 embeddings 和 labels
2. 加载或创建 split + noisy labels（缓存）
3. 构建 granular balls → 记录 gb_stats（球数、大小、purity）
4. 计算各方法的 reliability:
   - topk: k = GB 的平均球大小（matched）
   - kmeans: n_clusters = GB 的球数（matched）
   - gb: 直接用 group_ids
   - shuffled: shuffle_group_ids
   - random: random_group_ids
   - none: 全零
5. 跑全部方法:
   ce, sce, sel_topk, sel_kmeans, sel_gb, sel_shuffled, sel_random, sel_noball, clean_oracle
6. 保存结果到 reports/smoke_{dataset}_{noise}_{rate}_seed{s}/results.csv
   CSV 列: method, test_acc, selection_precision, noise_exclusion_recall
7. 保存 gb_stats 到同目录 gb_stats.json
8. 打印格式化报告

报告必须包含:
- 各方法 test_acc（排序）
- clean_oracle - ce = headroom
- sel_gb - sel_topk = GB vs kNN delta
- sel_gb - sel_kmeans = GB vs kmeans delta
- sel_gb - sel_shuffled = 结构有效性
- GB 统计：球数、平均/最小/最大球大小、平均 purity

7. 冒烟验收标准

  • 所有 9 个方法都跑完，results.csv 有 9 行。
  • clean_oracle test_acc 明显高于 ce test_acc（headroom ≥ 3pp 为佳）。
  • sel_gb 的 selection_precision > 0.6（random clean rate for symmetric 0.4）。
  • sel_shuffled 和 sel_random 的 test_acc 低于 sel_gb。
如果 headroom < 2pp：说明 GraphCL 表示太好或噪声影响太小，需要排查。记录数字但继续。
如果 sel_gb < sel_topk：正常，seed0 不做结论，但如实记录。

8. 护栏（硬约束，违反即作废）

  1. Clean labels 只用于 val/test 评估 + selection_precision/noise_exclusion 事后诊断。绝不进入 reliability / selection / loss / target / 构球。
  2. 所有 sel_* 方法同 frozen embedding、同 split、同噪声实例、同 schedule、同 epochs/lr/patience。
  3. 不 threshold-shopping：所有超参来自 configs/default.yaml，不为通过冒烟调参。
  4. 不做 hard same-ball positive、prototype pulling、hard inter-ball negative。
  5. test 一律 test_acc_at_best_val；数字来自 CSV。
  6. 不碰 pair noise、encoder coupling、5 seed gate——那是下一阶段。
  7. 新增代码必须补单测并保持全绿。
